### 2. 后端代码

#### app.py（Flask后端）

```python
from flask import Flask, request, jsonify, send_from_directory
from flask_cors import CORS
import sqlite3
import hashlib
import time
from datetime import datetime

app = Flask(__name__, static_folder='.')
CORS(app)

# 数据库初始化
def init_db():
    conn = sqlite3.connect('fission.db')
    c = conn.cursor()

    # 用户表
    c.execute('''CREATE TABLE IF NOT EXISTS users
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  name TEXT NOT NULL,
                  phone TEXT UNIQUE NOT NULL,
                  coupon_code TEXT NOT NULL,
                  parent_id INTEGER DEFAULT 0,
                  invite_count INTEGER DEFAULT 0,
                  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)''')

    # 邀请关系表
    c.execute('''CREATE TABLE IF NOT EXISTS invites
                 (id INTEGER PRIMARY KEY AUTOINCREMENT,
                  inviter_id INTEGER NOT NULL,
                  invitee_id INTEGER NOT NULL,
                  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP)''')

    conn.commit()
    conn.close()

init_db()

# 生成优惠码
def generate_coupon_code():
    return hashlib.md5(str(time.time()).encode()).hexdigest()[:8].upper()

# 提交用户信息
@app.route('/api/submit', methods=['POST'])
def submit():
    data = request.json
    name = data.get('name')
    phone = data.get('phone')
    pid = data.get('pid', '')

    if not name or not phone:
        return jsonify({'success': False, 'message': '参数错误'})

    conn = sqlite3.connect('fission.db')
    c = conn.cursor()

    try:
        # 检查手机号是否已存在
        c.execute('SELECT id FROM users WHERE phone = ?', (phone,))
        existing = c.fetchone()

        if existing:
            return jsonify({'success': False, 'message': '该手机号已领取过优惠券'})

        # 生成优惠码
        coupon_code = generate_coupon_code()

        # 插入用户
        parent_id = int(pid) if pid and pid.isdigit() else 0
        c.execute('INSERT INTO users (name, phone, coupon_code, parent_id) VALUES (?, ?, ?, ?)',
                  (name, phone, coupon_code, parent_id))
        user_id = c.lastrowid

        # 如果有推荐人，记录邀请关系并更新推荐人的邀请数
        if parent_id > 0:
            c.execute('INSERT INTO invites (inviter_id, invitee_id) VALUES (?, ?)',
                      (parent_id, user_id))
            c.execute('UPDATE users SET invite_count = invite_count + 1 WHERE id = ?',
                      (parent_id,))

        conn.commit()

        return jsonify({
            'success': True,
            'user_id': user_id,
            'coupon_code': coupon_code
        })

    except Exception as e:
        conn.rollback()
        return jsonify({'success': False, 'message': str(e)})
    finally:
        conn.close()

# 获取用户信息
@app.route('/api/user/<int:user_id>', methods=['GET'])
def get_user(user_id):
    conn = sqlite3.connect('fission.db')
    c = conn.cursor()

    c.execute('SELECT coupon_code, invite_count FROM users WHERE id = ?', (user_id,))
    user = c.fetchone()
    conn.close()

    if user:
        return jsonify({
            'success': True,
            'coupon_code': user[0],
            'invite_count': user[1]
        })
    else:
        return jsonify({'success': False, 'message': '用户不存在'})

# 静态文件服务
@app.route('/')
def index():
    return send_from_directory('.', 'index.html')

@app.route('/<path:path>')
def static_files(path):
    return send_from_directory('.', path)

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)
```
