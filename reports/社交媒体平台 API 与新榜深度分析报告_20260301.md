### 4.1 使用新榜 API 实现监控

#### 接入流程

```
1. 注册新榜账号
   ↓
2. 开通专业版/企业版
   ↓
3. 获取 API Key
   ↓
4. 调用 API 获取数据
   ↓
5. 数据处理和存储
   ↓
6. 生成监控报告
```

#### 代码示例

```python
import requests

# 配置
API_KEY = "your_api_key"
BASE_URL = "https://api.newrank.cn"

# 获取公众号文章列表
def get_wechat_articles(account_id, limit=10):
    url = f"{BASE_URL}/api/article/list"
    params = {
        "account_id": account_id,
        "limit": limit
    }
    headers = {
        "Authorization": f"Bearer {API_KEY}"
    }
    response = requests.get(url, params=params, headers=headers)
    return response.json()

# 获取抖音视频列表
def get_douyin_videos(author_id, limit=10):
    url = f"{BASE_URL}/api/video/list"
    params = {
        "author_id": author_id,
        "limit": limit
    }
    headers = {
        "Authorization": f"Bearer {API_KEY}"
    }
    response = requests.get(url, params=params, headers=headers)
    return response.json()

# 获取小红书笔记列表
def get_xiaohongshu_notes(user_id, limit=10):
    url = f"{BASE_URL}/api/note/list"
    params = {
        "user_id": user_id,
        "limit": limit
    }
    headers = {
        "Authorization": f"Bearer {API_KEY}"
    }
    response = requests.get(url, params=params, headers=headers)
    return response.json()
```
